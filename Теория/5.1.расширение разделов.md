# Проблема — No space left on device

Диск manager2 заполнен. Это та же проблема что была в Part 1 — базовый образ маленький.

## Расширяем диск на manager2

```bash
# Расширяем через qemu-guest-agent
guest_run() {
  local domain="$1"
  local cmd="$2"
  local b64=$(echo -n "$cmd" | base64 -w0)
  local pid=$(sudo virsh qemu-agent-command "$domain" \
    "{\"execute\":\"guest-exec\",\"arguments\":{\"path\":\"/bin/sh\",\"arg\":[\"-c\",\"echo $b64 | base64 -d | sh\"],\"capture-output\":true}}" \
    | python3 -c 'import sys,json; print(json.load(sys.stdin)["return"]["pid"])')
  sleep 3
  sudo virsh qemu-agent-command "$domain" \
    "{\"execute\":\"guest-exec-status\",\"arguments\":{\"pid\":$pid}}" \
    | python3 -c '
import sys, json, base64
r = json.load(sys.stdin)["return"]
if r.get("out-data"): print(base64.b64decode(r["out-data"]).decode())
if r.get("err-data"): print("STDERR:", base64.b64decode(r["err-data"]).decode())
print("exitcode:", r.get("exitcode", "?"))
'
}

# Проверяем место на всех 4 ВМ
guest_run ansible02_manager2 "df -h /"
guest_run ansible02_consul_server "df -h /"
guest_run ansible02_api "df -h /"
guest_run ansible02_db "df -h /"
```

## Расширяем разделы

```bash
# Расширяем на всех 4 (на будущее)
for vm in ansible02_manager2 ansible02_consul_server ansible02_api ansible02_db; do
  echo "=== $vm ==="
  guest_run "$vm" "growpart /dev/vda 1 2>/dev/null; resize2fs /dev/vda1 2>/dev/null; df -h /"
done
```

## После расширения — ставим Ansible

```bash

ssh manager2 "which ansible && ansible --version"
```
